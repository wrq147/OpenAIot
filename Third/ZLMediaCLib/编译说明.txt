windows下convert.c编译命令
cl /LD /O2 /W4 convert.c /Fe:convert.dll

windows下ff_h264_encoder.c编译命令
cl /LD ff_h264_encoder.c /I include /link avcodec.lib avutil.lib swscale.lib /LIBPATH:lib /OUT:ff_h264_encoder.dll

linux下convert.c编译命令
gcc -shared -fPIC -O2 -Wall convert.c -o libconvert.so

linux下ff_h264_encoder.c编译命令
Ubuntu / Debian 系列   apt install -y libavcodec-dev libavutil-dev libswscale-dev
CentOS / RHEL / Fedora   yum install -y ffmpeg-devel
gcc -shared -fPIC -o libff_h264_encoder.so ff_h264_encoder.c -lavcodec -lavutil -lswscale
